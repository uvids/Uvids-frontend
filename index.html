async function getVideo() { var urlInput = document.getElementById('videoUrl').value; var errorDiv = document.getElementById('errorDiv'); var btn = document.getElementById('mainBtn'); var btnText = document.getElementById('btnText'); var btnLoader = document.getElementById('btnLoader'); var resultArea = document.getElementById('resultArea'); if (!urlInput) { errorDiv.style.display = 'block'; errorDiv.innerText = "Please enter a valid URL."; return; } errorDiv.style.display = 'none'; btn.disabled = true; btnText.style.display = 'none'; btnLoader.style.display = 'block'; resultArea.style.display = 'none'; var myApiUrl = 'https://nebula-api-6cc6.onrender.com/download'; var maxRetries = 3; var delay = 5000; // 5 seconds wait for (let i = 0; i < maxRetries; i++) { try { var response = await fetch(myApiUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ url: urlInput }) }); var data = await response.json(); if (data.status === "success") { var thumbImg = document.getElementById('vidThumb'); if (data.thumbnail) { thumbImg.src = data.thumbnail; thumbImg.style.display = 'block'; } else { thumbImg.style.display = 'none'; } document.getElementById('vidTitle').innerText = data.title || "Video Found"; currentHdUrl = data.hd_url || ""; currentSdUrl = data.sd_url || ""; currentAudioUrl = data.audio_url || ""; document.getElementById('btnHd').style.display = currentHdUrl ? 'flex' : 'none'; document.getElementById('btnSd').style.display = currentSdUrl ? 'flex' : 'none'; document.getElementById('btnMp3').style.display = currentAudioUrl ? 'flex' : 'none'; document.getElementById('btnHd').onclick = function(e) { e.preventDefault(); triggerDownload(currentHdUrl); }; document.getElementById('btnSd').onclick = function(e) { e.preventDefault(); triggerDownload(currentSdUrl); }; document.getElementById('btnMp3').onclick = function(e) { e.preventDefault(); triggerDownload(currentAudioUrl); }; resultArea.style.display = 'block'; btn.disabled = false; btnText.style.display = 'block'; btnLoader.style.display = 'none'; return; // Success! Loop se bahar nikal jao } else { // Agar API error de rahi hai errorDiv.style.display = 'block'; errorDiv.innerText = "Error: " + (data.detail || "Video not found."); break; // Retry mat karo agar link hi galat hai } } catch (e) { // Agar network error aaya (Server so raha hai), toh retry karo if (i < maxRetries - 1) { btnLoader.style.display = 'none'; btnText.innerText = `Waking server... (Try ${i+1}/${maxRetries})`; btnText.style.display = 'block'; await new Promise(r => setTimeout(r, delay)); // Wait karo btnText.style.display = 'none'; btnLoader.style.display = 'block'; } else { errorDiv.style.display = 'block'; errorDiv.innerText = "Server is taking too long. Please try again later."; } } } btn.disabled = false; btnText.style.display = 'block'; btnLoader.style.display = 'none'; }
